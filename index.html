import { GoogleGenAI, Type } from '@google/genai'; import * as fs from 'fs'; import dotenv from 'dotenv'; dotenv.config(); const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY }); /** * 1. Vision API: 画像（レシート/冷蔵庫）から食材データと優先度タグを抽出 */ export async function analyzeFridgeImage(imagePath: string, mimeType: string = 'image/jpeg') { const imageBuffer = fs.readFileSync(imagePath); const base64Image = imageBuffer.toString('base64'); const today = new Date().toISOString().split('T')[0]; // YYYY-MM-DD const systemInstruction = ` # Role & Task あなたは画像認識とデータ抽出のスペシャリストです。 入力された「冷蔵庫内の写真」または「スーパーのレシート画像」から、読み取れるすべての食材・購入品を抽出し、指定のJSONフォーマットで出力してください。 # Processing Rules 1. 画像種別の判定: レシート画像の場合はOCR処理、冷蔵庫内の写真の場合は物体認識を行ってください。 2. 賞味期限/消費期限の推定: 明記されている日付を採用し、不明な場合は一般的な保存期間に基づき本日(${today})から逆算してください。 3. 優先度タグ(priority_tag)の付与基準: - 【超優先】: 残り日数 0日以下（本日中）または傷みやすい生の肉・魚類 - 【優先】 : 残り日数 1日（明日まで） - 【注意】 : 残り日数 2日 - 【通常】 : 残り日数 3日以上、または日持ちする調味料・乾物類 `; // 出力構造の定義 (JSON Schema) const responseSchema = { type: Type.OBJECT, properties: { source_type: { type: Type.STRING, enum: ['receipt', 'fridge_photo'] }, processed_at: { type: Type.STRING }, items: { type: Type.ARRAY, items: { type: Type.OBJECT, properties: { item_id: { type: Type.STRING }, name: { type: Type.STRING }, quantity: { type: Type.STRING }, expiration_date: { type: Type.STRING }, days_left: { type: Type.INTEGER }, priority_tag: { type: Type.STRING, enum: ['【超優先】', '【優先】', '【注意】', '【通常】'] }, category: { type: Type.STRING }, notes: { type: Type.STRING } }, required: ['item_id', 'name', 'priority_tag', 'days_left'] } } }, required: ['source_type', 'items'] }; const response = await ai.models.generateContent({ model: 'gemini-2.5-flash', contents: [ { role: 'user', parts: [ { inlineData: { mimeType, data: base64Image } }, { text: `本日日付は ${today} です。画像に含まれる食材を解析してください。` } ] } ], config: { systemInstruction, responseMimeType: 'application/json', responseSchema } }); return JSON.parse(response.text!); } /** * 2. Recipe API: 食材リストからレシピおよび余った食材の保存テクニックを生成 */ export async function generateRecipesAndPreservation(ingredientItems: any[], userConstraints: string = '') { const systemInstruction = ` # Role & Goal あなたは家庭の「食品ロスゼロ」と「献立決定のストレス軽減」を達成する高度な料理アドバイザーAIです。 入力される食材の「優先度タグ」を厳密に評価し、賞味期限切れ間近の食材を使い切るレシピと、余った食材の保存テクニックを提案してください。 # Rules 1. 【超優先】タグの食材は原則すべて使い切るレシピにしてください。 2. レシピで使い切れず余る【超優先】【優先】食材がある場合は、長持ち保存・冷凍テクニック(preservation_guides)を提案してください。 3. アレルギー/NG食材は絶対に含めないでください。 `; const responseSchema = { type: Type.OBJECT, properties: { recipes: { type: Type.ARRAY, items: { type: Type.OBJECT, properties: { recipe_id: { type: Type.STRING }, title: { type: Type.STRING }, cooking_time_minutes: { type: Type.INTEGER }, servings: { type: Type.INTEGER }, priority_ingredients_cleared: { type: Type.ARRAY, items: { type: Type.STRING } }, other_ingredients_used: { type: Type.ARRAY, items: { type: Type.STRING } }, seasonings: { type: Type.ARRAY, items: { type: Type.STRING } }, steps: { type: Type.ARRAY, items: { type: Type.STRING } }, safety_and_tips: { type: Type.STRING } }, required: ['recipe_id', 'title', 'steps', 'priority_ingredients_cleared'] } }, preservation_guides: { type: Type.ARRAY, items: { type: Type.OBJECT, properties: { ingredient_name: { type: Type.STRING }, method_type: { type: Type.STRING }, title: { type: Type.STRING }, instructions: { type: Type.ARRAY, items: { type: Type.STRING } }, shelf_life: { type: Type.STRING }, thawing_tip: { type: Type.STRING } }, required: ['ingredient_name', 'title', 'instructions', 'shelf_life'] } } }, required: ['recipes', 'preservation_guides'] }; const formattedIngredients = ingredientItems .map(item => `- ${item.priority_tag} ${item.name} (${item.quantity || '適量'})`) .join('\n'); const userPrompt = ` - 抽出された食材リスト: ${formattedIngredients} - ユーザーからの条件・制限事項: ${userConstraints} `; const response = await ai.models.generateContent({ model: 'gemini-2.5-flash', contents: userPrompt, config: { systemInstruction, responseMimeType: 'application/json', responseSchema } }); return JSON.parse(response.text!); }
